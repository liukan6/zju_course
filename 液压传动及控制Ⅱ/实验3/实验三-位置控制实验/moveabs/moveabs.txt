function moveabs(filename,sheet,color,N1,N2,N3,N4)
Y1 = xlsread(filename,sheet);
%% 调试用
% Y1 = xlsread('test1.dbf','test1');
% color='r'                          
% N1=500;
% N2=500;
% N3=500;
% N4=500;
%Y1 = Y(0:end,1:7);
%% 参数设置
Target_p=1;         %Target Position油缸目标位置所在列
Actual_P=2;         %Actual Position油缸实际位置所在列
Interpolated_P=3;   %Interpolated Position油缸插值位置所在列
Interpolated_V=4;   %Interpolated Velocity油缸插值速度所在列
ValueA=5;           %用户自定义数据所在列
IAC_Apre=6;         %IAC_Apre所在列
IAC_Pos_following_error=7;     %IAC_Pos_following error所在列
IAC_valve_cmd=8;    %IAC_valve_cmd所在列
Fs=500;             %控制器控制周期2ms
[m,n]=size(Y1);         %采集数据的维度
color1=strcat(color,':');
color2=strcat(color,'--');
color3=strcat(color,'-.');
color4=strcat(color,'-');
color5=strcat(color,'--');
%% 获取目标位置信号的改变点
index = [];             %保存数据转折的行号
j=1;
Target_column=Interpolated_P;
for i=1:m-2
    first = Y1(i,Target_column);
    second =  Y1(i+1,Target_column);
    third = Y1(i+2,Target_column);
    if (first == second && second ~= third) || (first ~= second && second == third) 
        index(j)=i+1;
        j=j+1;
    end
end
%% 画图
t1=0:1/Fs:(index(2)-index(1)+N1+N2)/Fs;
t2=(index(2)-index(1)+N1+N2)/Fs:1/Fs:(index(2)-index(1)+index(4)-index(3)+N1+N2+N3+N4)/Fs;

yyaxis left
plot(t1,Y1(index(1)-N1:index(2)+N2,Target_p),color1)
hold on
plot(t1,Y1(index(1)-N1:index(2)+N2,Actual_P),color1)
hold on
plot(t1,Y1(index(1)-N1:index(2)+N2,Interpolated_P),color2)
hold on
plot(t1,Y1(index(1)-N1:index(2)+N2,Interpolated_V),color3)
hold on

plot(t2,Y1(index(3)-N3:index(4)+N4,Target_p),color1)
hold on
plot(t2,Y1(index(3)-N3:index(4)+N4,Actual_P),color1)
hold on
plot(t2,Y1(index(3)-N3:index(4)+N4,Interpolated_P),color2)
plot(t2,Y1(index(3)-N3:index(4)+N4,Interpolated_V),color3)
hold on
xline(N1/Fs);
xline((N1+index(2)-index(1))/Fs);
xline((N1+N2+index(2)-index(1))/Fs);
xline((N1+N2+index(2)-index(1)+N3)/Fs);
xline((N1+N2+index(2)-index(1)+N3+index(4)-index(3))/Fs);
xline((N1+N2+index(2)-index(1)+N3+index(4)-index(3)+N4)/Fs);
yline(0);
xlabel('时间(s)');
ylabel('位移（mm）,速度（mm/s）');

%%
yyaxis right
plot(t1,Y1(index(1)-N1:index(2)+N2,Interpolated_P)-Y1(index(1)-N1:index(2)+N2,Actual_P),color4);
hold on
plot(t1,Y1(index(1)-N1:index(2)+N2,IAC_Pos_following_error),color5);
hold on
plot(t2,Y1(index(3)-N3:index(4)+N4,Interpolated_P)-Y1(index(3)-N3:index(4)+N4,Actual_P),color4);
hold on
plot(t2,Y1(index(3)-N3:index(4)+N4,IAC_Pos_following_error),color5);
hold on
yline(0);
xlabel('时间(s)');
ylabel('位置误差（mm）');
